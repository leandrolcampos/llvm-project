cmake_minimum_required(VERSION 3.30.4)
project(GpuLibmBenchRunner LANGUAGES CXX CUDA)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(deps "${REPO}/third-party/nvbench-deps")
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Offline vendored dependencies" FORCE)
set(FETCHCONTENT_SOURCE_DIR_RAPIDS-CMAKE "${deps}/rapids-cmake" CACHE PATH "" FORCE)
set(CPM_DOWNLOAD_LOCATION "${deps}/CPM.cmake" CACHE FILEPATH "" FORCE)
set(CPM_nlohmann_json_SOURCE "${deps}/json")
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${deps}/fmt" "${CMAKE_CURRENT_BINARY_DIR}/fmt" EXCLUDE_FROM_ALL)
set(NVBench_ENABLE_INSTALL_RULES OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_HEADER_TESTING OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_DEVICE_TESTING OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_EXECUTABLES OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_CUPTI OFF CACHE BOOL "" FORCE)
set(NVBench_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
# NVBench is vendored, not a checkout. Do not attribute LLVM's HEAD to it.
set(CMAKE_DISABLE_FIND_PACKAGE_Git TRUE)
add_subdirectory("${REPO}/third-party/nvbench" "${CMAKE_CURRENT_BINARY_DIR}/nvbench" EXCLUDE_FROM_ALL)
find_package(CUDAToolkit REQUIRED)
add_executable(gpu-libm-bench main.cu)
target_link_libraries(gpu-libm-bench PRIVATE nvbench::nvbench CUDA::cuda_driver nlohmann_json::nlohmann_json)
target_compile_features(gpu-libm-bench PRIVATE cuda_std_17)
target_compile_definitions(gpu-libm-bench PRIVATE
  BENCH_DEVICE_DIR="${BENCH_DEVICE_DIR}" BENCH_REGISTRY="${BENCH_REGISTRY}")
set_target_properties(gpu-libm-bench PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
